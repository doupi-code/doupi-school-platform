#### 2.3 同步脚本
```javascript
// scripts/sync-shared.js
const fs = require('fs');
const path = require('path');

const shared = './doupi-mp-shared';
const targets = ['./doupi-app-client', './doupi-app-admin'];

// 递归复制共享文件到两个目标
function syncFiles(srcDir, relativePath = '') {
  const items = fs.readdirSync(path.join(srcDir, relativePath));
  for (const item of items) {
    const rel = path.join(relativePath, item);
    const src = path.join(srcDir, rel);
    if (fs.statSync(src).isDirectory()) {
      syncFiles(srcDir, rel);
    } else {
      for (const target of targets) {
        const dest = path.join(target, rel);
        fs.mkdirSync(path.dirname(dest), { recursive: true });
        fs.copyFileSync(src, dest);
      }
    }
  }
}

syncFiles(shared);
console.log('同步完成！');
```
